# Plan: Auth Module — как в Spring Security

## Принцип

Как в Research_Keycloack: **всё в сессии**. Никаких отдельных Redis ключей для OAuth flow.

---

## Требования

1. **Токен в прокси** — передача Bearer token при проксировании на protected service
2. **Backchannel logout** — хранение маппинга user→sessions, удаление ВСЕХ сессий
3. **Обычный logout** — удаление ТОЛЬКО текущей сессии
4. **Refresh token endpoint** — `POST /api/auth/refresh`
5. **Динамические URL Keycloak** — все вызовы через `${config.keycloak.issuer}/protocol/openid-connect/...`

---

## Исправления

1. **RedisService** — injectable, один клиент через DI
2. **session.id** — не `session.req?.sessionID`
3. **TTL маппинга** — обновлять при refresh
4. **Logout CSRF** — сменить GET на POST
5. **Ошибка прокси** — логировать, не утекать детали
6. **Logout поведение** — обычный = только текущая сессия, backchannel = все сессии
7. **Audience check** — проверка `aud` в backchannel logout

```typescript
import { Injectable, OnModuleDestroy } from '@nestjs/common';
import { createClient, RedisClientType } from 'redis';
import { Logger } from '../utils/logger';

@Injectable()
export class RedisService implements OnModuleDestroy {
  public readonly client: RedisClientType;

  constructor() {
    this.client = createClient({
      url: `redis://${process.env.REDIS_HOST || 'localhost'}:${process.env.REDIS_PORT || '6379'}`,
      password: process.env.REDIS_PASSWORD || undefined,
    });

    this.client.on('error', (err) => {
      Logger.error('Redis', `Client error: ${err.message}`);
    });

    this.client.on('connect', () => {
      Logger.info('Redis', 'Connected');
    });

    this.client.connect().catch((err) => {
      Logger.error('Redis', `Connection failed: ${err.message}`);
    });
  }

  async onModuleDestroy() {
    await this.client.disconnect();
  }
}
```

```typescript
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { config } from './configs/configuration';
import { RedisService } from './services/redis.service';
import * as cookieParser from 'cookie-parser';
import * as session from 'express-session';
import RedisStore from 'connect-redis';
import { Logger } from './utils/logger';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  const redisService = app.get(RedisService);

  app.use(cookieParser());

  app.use(
    session({
      store: new RedisStore({ 
        client: redisService.client,
        prefix: config.session.prefix,
      }),
      secret: process.env.SESSION_SECRET || 'your-secret-key',
      resave: false,
      saveUninitialized: false,
      cookie: {
        httpOnly: true,
        secure: false,
        sameSite: 'lax',
        maxAge: config.session.ttl * 1000,
      },
    }),
  );

  app.enableCors({
    origin: config.frontendUrl,
    methods: 'GET,HEAD,PUT,PATCH,POST,DELETE,OPTIONS',
    credentials: true,
  });

  await app.listen(config.port);
  Logger.info('App', `BFF Application is running on: http://localhost:${config.port}`);
}

bootstrap();
```

### `bff/src/services/auth.service.ts`

```typescript
import { Injectable, UnauthorizedException } from '@nestjs/common';
import { config } from '@configs/configuration';
import { RedisService } from './redis.service';
import { RedisClientType } from 'redis';
import axios from 'axios';
import * as crypto from 'crypto';
import * as querystring from 'querystring';
import { Logger } from '../utils/logger';
import { Session } from 'express-session';

@Injectable()
export class AuthService {
  private readonly redis: RedisClientType;

  constructor(private redisService: RedisService) {
    this.redis = redisService.client;
  }

  async getAuthorizationUrl(session: Session): Promise<{ url: string }> {
    const codeVerifier = crypto.randomBytes(32).toString('hex');
    const codeChallenge = crypto.createHash('sha256').update(codeVerifier).digest().toString('base64url');
    const state = crypto.randomBytes(16).toString('hex');

    session.oauth = { state, codeVerifier };
    session.csrfToken = crypto.randomBytes(32).toString('hex');

    const params = querystring.stringify({
      client_id: config.keycloak.clientId,
      response_type: 'code',
      redirect_uri: config.keycloak.redirectUri,
      scope: config.keycloak.scope,
      state,
      code_challenge: codeChallenge,
      code_challenge_method: 'S256',
    });

    return { url: `${config.keycloak.issuer}/protocol/openid-connect/auth?${params}` };
  }

  async handleCallback(code: string, state: string, session: Session) {
    if (!session.oauth) throw new UnauthorizedException('No pending auth');
    if (session.oauth.state !== state) throw new UnauthorizedException('State mismatch');

    const postData = querystring.stringify({
      grant_type: 'authorization_code',
      client_id: config.keycloak.clientId,
      client_secret: config.keycloak.clientSecret,
      redirect_uri: config.keycloak.redirectUri,
      code,
      code_verifier: session.oauth.codeVerifier,
    });

    try {
      const { data: tokenSet } = await axios.post(
        `${config.keycloak.issuer}/protocol/openid-connect/token`,
        postData,
        { headers: { 'Content-Type': 'application/x-www-form-urlencoded' } },
      );

      const idTokenPayload = JSON.parse(Buffer.from(tokenSet.id_token.split('.')[1], 'base64').toString());
      const userInfo = {
        sub: idTokenPayload.sub,
        email: idTokenPayload.email,
        preferred_username: idTokenPayload.preferred_username,
        name: idTokenPayload.name,
      };

      delete session.oauth;
      session.accessToken = tokenSet.access_token;
      session.refreshToken = tokenSet.refresh_token;
      session.idToken = tokenSet.id_token;
      session.userInfo = userInfo;

      await this.registerSession(userInfo.sub, session.id);

      return { userInfo };
    } catch (error: any) {
      const status = error.response?.status || 'unknown';
      const kcError = error.response?.data?.error;
      Logger.error('AuthService', `Token request failed: ${status} ${kcError || error.message}`);
      throw new UnauthorizedException('Authentication failed');
    }
  }

  async refreshAccessToken(session: Session) {
    if (!session.refreshToken) {
      throw new UnauthorizedException('No refresh token available');
    }

    const postData = querystring.stringify({
      grant_type: 'refresh_token',
      client_id: config.keycloak.clientId,
      client_secret: config.keycloak.clientSecret,
      refresh_token: session.refreshToken,
    });

    try {
      const { data: tokenSet } = await axios.post(
        `${config.keycloak.issuer}/protocol/openid-connect/token`,
        postData,
        { headers: { 'Content-Type': 'application/x-www-form-urlencoded' } },
      );

      session.accessToken = tokenSet.access_token;
      session.refreshToken = tokenSet.refresh_token;

      if (session.userInfo?.sub) {
        await this.redis.expire(`user_sessions:${session.userInfo.sub}`, config.session.ttl);
      }

      return { accessToken: tokenSet.access_token, refreshToken: tokenSet.refresh_token };
    } catch (error: any) {
      const status = error.response?.status || 'unknown';
      const kcError = error.response?.data?.error;
      const isInvalidGrant = status === 400 && kcError === 'invalid_grant';

      if (isInvalidGrant) {
        throw error;
      }

      throw new UnauthorizedException('Token refresh failed');
    }
  }

  getLogoutUrl(idToken: string): string {
    const logoutUrl = new URL(`${config.keycloak.issuer}/protocol/openid-connect/logout`);
    logoutUrl.searchParams.append('id_token_hint', idToken);
    logoutUrl.searchParams.append('post_logout_redirect_uri', `${config.frontendUrl}/home`);
    return logoutUrl.toString();
  }

  private async registerSession(userId: string, sessionId: string) {
    await this.redis.sAdd(`user_sessions:${userId}`, sessionId);
    await this.redis.expire(`user_sessions:${userId}`, config.session.ttl);
  }

  async unregisterSession(userId: string, sessionId: string) {
    await this.redis.sRem(`user_sessions:${userId}`, sessionId);
  }

  async destroyUserSessions(userId: string) {
    const sessionIds = await this.redis.sMembers(`user_sessions:${userId}`);
    if (sessionIds.length === 0) return;

    for (const sid of sessionIds) {
      await this.redis.del(`${config.session.prefix}${sid}`);
    }
    await this.redis.del(`user_sessions:${userId}`);
    Logger.info('AuthService', `Destroyed ${sessionIds.length} sessions for user: ${userId}`);
  }
}
```

### `bff/src/controllers/auth.controller.ts`

```typescript
import { Controller, Get, Post, Query, Req, Res, UseGuards, Header } from '@nestjs/common';
import { Response, Request } from 'express';
import { AuthService } from '@services/auth.service';
import { SessionGuard } from '@guards/session.guard';
import { CsrfGuard } from '@guards/csrf.guard';
import { config } from '@configs/configuration';
import { Logger } from '../utils/logger';
import { createRemoteJWKSet, jwtVerify } from 'jose';

@Controller()
export class AuthController {
  constructor(private authService: AuthService) {}

  @Get('login')
  async login(@Req() req: any, @Res() res: Response) {
    const { url } = await this.authService.getAuthorizationUrl(req.session);
    res.redirect(url);
  }

  @Get('callback')
  async callback(
    @Query('code') code: string,
    @Query('state') state: string,
    @Req() req: any,
    @Res() res: Response,
  ) {
    try {
      await this.authService.handleCallback(code, state, req.session);

      Logger.info('AuthController', `Session completed`, {
        user: req.session.userInfo.preferred_username || req.session.userInfo.email,
      });
      res.redirect(config.frontendUrl + '/');
    } catch (e) {
      Logger.error('AuthController', `Callback error: ${e.message}`);
      res.redirect(`${config.frontendUrl}/login?error=auth_failed`);
    }
  }

  @Get('api/me')
  @UseGuards(SessionGuard)
  async me(@Req() req: any) {
    return {
      user: req.session.userInfo,
      csrfToken: req.session.csrfToken,
    };
  }

  @Post('api/auth/refresh')
  @UseGuards(SessionGuard)
  async refresh(@Req() req: any, @Res() res: Response) {
    try {
      const tokens = await this.authService.refreshAccessToken(req.session);
      res.json(tokens);
    } catch (error) {
      res.status(401).json({ error: 'Token refresh failed' });
    }
  }

  @Post('logout')
  @UseGuards(SessionGuard, CsrfGuard)
  async logout(@Req() req: any, @Res() res: Response) {
    const idToken = req.session.idToken;
    const userId = req.session.userInfo?.sub;
    const sessionId = req.session.id;

    if (userId) {
      await this.authService.unregisterSession(userId, sessionId);
    }

    req.session.destroy((err: any) => {
      if (err) {
        Logger.error('AuthController', `Session destroy error: ${err.message}`);
      }
      res.clearCookie('connect.sid', { path: '/' });

      if (idToken) {
        const logoutUrl = this.authService.getLogoutUrl(idToken);
        res.redirect(logoutUrl);
      } else {
        res.redirect('/login');
      }
    });
  }

  @Post('api/auth/backchannel-logout')
  @Header('Cache-Control', 'no-store')
  async backchannelLogout(@Req() req: Request, @Res() res: Response) {
    const logoutToken = req.body?.logout_token;

    if (!logoutToken) {
      Logger.warn('AuthController', 'Backchannel logout: no logout_token in body');
      res.status(400).send('Missing logout_token');
      return;
    }

    try {
      const JWKS = createRemoteJWKSet(new URL(`${config.keycloak.issuer}/protocol/openid-connect/certs`));
      const { payload } = await jwtVerify(logoutToken, JWKS, {
        issuer: config.keycloak.issuer,
        audience: config.keycloak.clientId,
      });

      const sub = payload.sub as string;

      Logger.info('AuthController', 'Backchannel logout received', { sub });

      await this.authService.destroyUserSessions(sub);

      res.status(200).send('OK');
    } catch (error) {
      Logger.error('AuthController', `Backchannel logout validation failed: ${error.message}`);
      res.status(401).send('Invalid logout token');
    }
  }
}
```

```typescript
import { Injectable, CanActivate, ExecutionContext, UnauthorizedException } from '@nestjs/common';
import { Logger } from '../utils/logger';

@Injectable()
export class SessionGuard implements CanActivate {
  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const session = request.session;

    if (!session || !session.accessToken) {
      Logger.warn('SessionGuard', `No valid session`, {
        method: request.method,
        path: request.path,
      });
      throw new UnauthorizedException('No session');
    }

    return true;
  }
}
```

### `bff/src/guards/csrf.guard.ts`

```typescript
import { Injectable, CanActivate, ExecutionContext, ForbiddenException } from '@nestjs/common';

@Injectable()
export class CsrfGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const csrfHeader = request.headers['x-csrf-token'];
    const session = request.session;

    if (!session || !csrfHeader || csrfHeader !== session.csrfToken) {
      throw new ForbiddenException('Invalid or missing CSRF token');
    }

    return true;
  }
}
```

```typescript
import { AxiosRequestConfig, AxiosResponse } from 'axios';
import { axiosInstance } from './axios';

export class HttpClient {
  async get<TResponse, TParams>(
    url: string,
    params?: TParams,
    headers?: Record<string, string>,
  ): Promise<TResponse> {
    const config: AxiosRequestConfig<TParams> = { params, headers };
    const response: AxiosResponse<TResponse> = await axiosInstance.get(url, config);
    return response.data;
  }

  async post<TResponse, TRequest, TParams>(
    url: string,
    data: TRequest,
    params?: TParams,
    headers?: Record<string, string>,
  ): Promise<TResponse> {
    const config: AxiosRequestConfig<TRequest> = { params, headers };
    const response: AxiosResponse<TResponse> = await axiosInstance.post(url, data, config);
    return response.data;
  }
}

export const httpClient = new HttpClient();
```

### `bff/src/controllers/protected_service.controller.ts`

```typescript
import { Controller, Get, Post, Req, Res, UseGuards, Param } from '@nestjs/common';
import { Response } from 'express';
import { SessionGuard } from '@guards/session.guard';
import { CsrfGuard } from '@guards/csrf.guard';
import { httpClient } from '../http/httpClient';
import { config } from '@configs/configuration';
import { Logger } from '../utils/logger';

@Controller('api/service')
export class ProtectedServiceController {
  @Get()
  @UseGuards(SessionGuard)
  async proxyGetRoot(@Req() req: any, @Res() res: Response) {
    return this.proxyGet('', req, res);
  }

  @Get('*')
  @UseGuards(SessionGuard)
  async proxyGet(@Param('0') path: string, @Req() req: any, @Res() res: Response) {
    const targetUrl = `${config.protectedServiceUrl}/${path}`;
    const headers = { Authorization: `Bearer ${req.session.accessToken}` };

    try {
      const data = await httpClient.get(targetUrl, undefined, headers);
      res.json(data);
    } catch (error) {
      Logger.error('Proxy', `GET ${targetUrl} failed: ${error.message}`);
      const status = error.response?.status || 500;
      res.status(status).json({ error: 'Upstream request failed' });
    }
  }

  @Post('*')
  @UseGuards(SessionGuard, CsrfGuard)
  async proxyPost(@Param('0') path: string, @Req() req: any, @Res() res: Response) {
    const targetUrl = `${config.protectedServiceUrl}/${path}`;
    const headers = { Authorization: `Bearer ${req.session.accessToken}` };

    try {
      const data = await httpClient.post(targetUrl, req.body, undefined, headers);
      res.json(data);
    } catch (error) {
      Logger.error('Proxy', `POST ${targetUrl} failed: ${error.message}`);
      const status = error.response?.status || 500;
      res.status(status).json({ error: 'Upstream request failed' });
    }
  }
}
```

## Верификация

1. `GET /login` → редирект Keycloak
2. `GET /callback` → state валиден, токены получены
3. Подмена state → 401
4. `GET /api/me` → OK
5. `POST /api/auth/refresh` → новый access token
6. `GET /api/service/*` → Bearer token передаётся
7. `POST /logout` (с CSRF) → удалена только текущая сессия
8. `POST /api/auth/backchannel-logout` → удалены ВСЕ сессии пользователя
