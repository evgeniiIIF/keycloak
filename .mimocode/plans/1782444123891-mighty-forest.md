# Fix: infinite refresh loop when refresh_token expires

## Problem
`authService.refreshAccessToken` catches `invalid_grant` from Keycloak and wraps it in `UnauthorizedException`. Interceptor can't detect `invalid_grant` → session not deleted → infinite loop.

## Fix

### 1. `bff/src/services/auth.service.ts` — re-throw `invalid_grant` as original AxiosError
```ts
} catch (error: any) {
  const status = error.response?.status || 'UNKNOWN';
  const isInvalidGrant = status === 400 && error.response?.data?.error === 'invalid_grant';

  if (isInvalidGrant) {
    throw error;
  }

  Logger.error('AuthService', `Token refresh failed: ${status} ${error.message}`);
  throw new UnauthorizedException('Token refresh failed');
}
```

### 2. `bff/src/http/axios.ts` — catch refresh errors in handleInvalidAccessToken
```ts
async function handleInvalidAccessToken(error: AxiosError) {
  const originalRequest = error.config;
  const isRetry = (originalRequest as any)._retry;
  const sessionId = originalRequest?.headers?.['X-Session-ID'] as string;

  if (isRetry || !sessionId) return;

  (originalRequest as any)._retry = true;
  delete originalRequest.headers['X-Session-ID'];

  Logger.warn('HttpClient', `401 from ${originalRequest.url}, refreshing token...`);

  try {
    const newToken = await authService.refreshAccessToken(sessionId);
    originalRequest.headers.Authorization = `Bearer ${newToken}`;
    Logger.info('HttpClient', 'Token refreshed, retrying...');
    return axiosInstance(originalRequest);
  } catch (refreshError: any) {
    const isInvalidGrant = refreshError.response?.status === 400
      && refreshError.response?.data?.error === 'invalid_grant';

    if (isInvalidGrant) {
      Logger.error('HttpClient', `Refresh token expired (invalid_grant), session ${sessionId.slice(0, 8)}`);
      await sessionService.deleteSession(sessionId);
      throw new UnauthorizedException('Session expired');
    }

    throw refreshError;
  }
}
```
