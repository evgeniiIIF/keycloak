### 2. `bff/src/auth/guards/auth.guard.ts` (новый)

```typescript
import { Injectable, CanActivate, ExecutionContext, UnauthorizedException } from '@nestjs/common';
import { Request } from 'express';

@Injectable()
export class AuthGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const req = context.switchToHttp().getRequest<Request>();
    const s = req.session;
    if (!s?.accessToken || !s?.refreshToken || !s?.idToken || !s?.userInfo?.sub) {
      throw new UnauthorizedException('Not authenticated');
    }
    return true;
  }
}
```

### 3. `bff/src/auth/controllers/auth.controller.ts`

Добавить `@UseGuards(AuthGuard)` на me и logout, кастовать к `BffSession`:

```typescript
@UseGuards(AuthGuard)
@Get('api/me')
me(@Req() req: Request) {
  const session = req.session as BffSession;
  return { user: session.userInfo };
}

@UseGuards(AuthGuard)
@Post('logout')
async logout(@Req() req: Request, @Res() res: Response) {
  const session = req.session as BffSession;
  await this.authService.performLogout(session);
  res.clearCookie(config.session.cookieName, { path: '/', sameSite: 'lax', secure: config.isProduction });
  res.clearCookie('XSRF-TOKEN', { path: '/' });
  res.json({ logoutUrl });
}
```

### 4. `bff/src/auth/services/auth.service.ts`

Убрать defensive checks, удалить wrappers:

```typescript
async performLogout(session: BffSession): Promise<void> {
  await this.keycloak.revokeRefreshToken(session.refreshToken);  // без if
  await this.unregisterSession(session.userInfo.sub, session.id); // без if
  await this.destroySession(session);

private buildLogoutUrl(session: BffSession): string {
```

Удалить приватные методы `revokeRefreshToken` и `unregisterUserSession`.
