# Централизованная защита роутов в BFF

Deny-by-default: все роуты закрыты. Открываем только то, что нужно в конфиге.

## Файл 1: `bff/src/guards/route-security.config.ts`

```typescript
export interface RouteRule {
  /** Route pattern: '/exact', '/path/**', '/prefix/*' */
  pattern: string;
  /** HTTP method. Строка или массив. Опустить = все методы */
  method?: string | string[];
  /** Требовать валидную сессию + JWT */
  session?: boolean;
  /** Требовать CSRF double-submit */
  csrf?: boolean;
  /** Требовать роли из JWT */
  roles?: string[];
  /** 'any' = хотя бы одна роль (default), 'all' = все роли */
  roleMatch?: 'any' | 'all';
  /** Откуда читать роли: 'realm' = realm_access.roles (default),
   *  или clientId = resource_access[clientId].roles */
  roleSource?: string;
  /** Public — пропустить все проверки */
  public?: boolean;
}

/**
 * Правила проверяются СВЕРХУ ВНИЗ — первое совпадение побеждает.
 * Если роут не попал ни в одно правило → 403 (deny-by-default).
 */
export const ROUTE_SECURITY: RouteRule[] = [
  // ═══════════════════════════════════════════════════
  // PUBLIC — без авторизации
  // ═══════════════════════════════════════════════════
  { pattern: '/login', public: true },
  { pattern: '/callback', public: true },
  { pattern: '/health', public: true },
  { pattern: '/health/**', public: true },
  { pattern: '/api/auth/backchannel-logout', public: true },

  // ═══════════════════════════════════════════════════
  // SERVICE-A: public docs + auth + CSRF для мутаций
  // ═══════════════════════════════════════════════════
  { pattern: '/api/service-a/public/**', public: true },
  { pattern: '/api/service-a/**', method: ['POST', 'PUT', 'DELETE', 'PATCH'], session: true, csrf: true },
  { pattern: '/api/service-a/**', session: true },

  // ═══════════════════════════════════════════════════
  // SERVICE-B: admin-only
  // ═══════════════════════════════════════════════════
  { pattern: '/api/service-b/**', method: ['POST', 'PUT', 'DELETE'], session: true, csrf: true, roles: ['admin'] },
  { pattern: '/api/service-b/**', session: true, roles: ['admin'] },

  // ═══════════════════════════════════════════════════
  // SERVICE-C: client roles (resource_access)
  // ═══════════════════════════════════════════════════
  { pattern: '/api/service-c/**', method: ['POST', 'PUT', 'DELETE'], session: true, csrf: true, roles: ['writer'], roleSource: 'service-c' },
  { pattern: '/api/service-c/**', session: true, roles: ['reader'], roleSource: 'service-c' },

  // ═══════════════════════════════════════════════════
  // SERVICE-D: public docs + settings (all roles) + admin write
  // ═══════════════════════════════════════════════════
  { pattern: '/api/service-d/docs/**', public: true },
  { pattern: '/api/service-d/settings/**', session: true, roles: ['admin', 'editor'], roleMatch: 'all' },
  { pattern: '/api/service-d/**', method: ['POST', 'PUT', 'DELETE'], session: true, csrf: true, roles: ['admin'] },
  { pattern: '/api/service-d/**', session: true },

  // ═══════════════════════════════════════════════════
  // AUTH ENDPOINTS
  // ═══════════════════════════════════════════════════
  { pattern: '/api/me', session: true },
  { pattern: '/api/auth/refresh', session: true },

  // ═══════════════════════════════════════════════════
  // LOGOUT: только CSRF, сессия не обязательна
  // ═══════════════════════════════════════════════════
  { pattern: '/logout', csrf: true },
];
```

## Файл 2: `bff/src/guards/security.guard.ts`

```typescript
import {
  Injectable,
  CanActivate,
  ExecutionContext,
  ForbiddenException,
} from '@nestjs/common';
import { decodeJwt } from 'jose';
import { RouteRule, ROUTE_SECURITY } from './route-security.config';
import { SessionGuard } from './session.guard';
import { CsrfGuard } from './csrf.guard';
import { Logger } from '../utils/logger';

// ─── Паттерн-матчер ────────────────────────────────────────────────
function matchPath(pattern: string, path: string): boolean {
  if (pattern === path) return true;
  const pp = pattern.split('/').filter(Boolean);
  const tp = path.split('/').filter(Boolean);
  let pi = 0, ti = 0;
  while (pi < pp.length && ti < tp.length) {
    if (pp[pi] === '**') return true;
    if (pp[pi] === '*') { pi++; ti++; continue; }
    if (pp[pi] === tp[ti]) { pi++; ti++; continue; }
    return false;
  }
  return pi === pp.length && ti === tp.length;
}

function matchMethod(ruleMethod: string | string[] | undefined, reqMethod: string): boolean {
  if (!ruleMethod) return true;
  const methods = Array.isArray(ruleMethod) ? ruleMethod : [ruleMethod];
  return methods.includes(reqMethod);
}

function matchRoute(
  path: string,
  method: string,
  rules: RouteRule[],
): RouteRule | undefined {
  for (const rule of rules) {
    if (!matchPath(rule.pattern, path)) continue;
    if (!matchMethod(rule.method, method)) continue;
    return rule;
  }
  return undefined;
}

// ─── Guard ─────────────────────────────────────────────────────────
@Injectable()
export class SecurityGuard implements CanActivate {
  // Кэш: method:path → rule (или null если нет совпадения)
  private ruleCache = new Map<string, RouteRule | null>();

  constructor(
    private sessionGuard: SessionGuard,
    private csrfGuard: CsrfGuard,
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const req = context.switchToHttp().getRequest();
    const rule = this.getCachedRule(req.path, req.method);

    // DENY BY DEFAULT: нет правила = 403
    if (!rule) {
      Logger.warn('SecurityGuard', 'No rule — denied', {
        path: req.path,
        method: req.method,
      });
      throw new ForbiddenException('Route not configured');
    }

    // Public: пропускаем все проверки
    if (rule.public) return true;

    // 1) Session + JWT (делегируем в SessionGuard → бросает 401 если нет сессии)
    if (rule.session) {
      await this.sessionGuard.canActivate(context);
    }

    // 2) CSRF (делегируем в CsrfGuard → бросает 403 если невалидный)
    if (rule.csrf) {
      this.csrfGuard.canActivate(context);
    }

    // 3) Roles из конфига (403 если недостаточно прав)
    if (rule.roles?.length) {
      const session = req.session;
      if (!session?.accessToken) {
        throw new ForbiddenException('No session');
      }

      const payload = decodeJwt(session.accessToken) as any;

      const source = rule.roleSource || 'realm';
      const userRoles: string[] =
        source === 'realm'
          ? payload?.realm_access?.roles || []
          : payload?.resource_access?.[source]?.roles || [];

      const mode = rule.roleMatch || 'any';
      const ok =
        mode === 'all'
          ? rule.roles.every((r) => userRoles.includes(r))
          : rule.roles.some((r) => userRoles.includes(r));

      if (!ok) {
        Logger.warn('SecurityGuard', 'Insufficient roles', {
          required: rule.roles.join(','),
          mode,
          actual: userRoles.join(','),
        });
        throw new ForbiddenException('Insufficient roles');
      }
    }

    return true;
  }

  private getCachedRule(path: string, method: string): RouteRule | null {
    const key = `${method}:${path}`;
    if (this.ruleCache.has(key)) {
      return this.ruleCache.get(key)!;
    }
    const rule = matchRoute(path, method, ROUTE_SECURITY) || null;
    this.ruleCache.set(key, rule);
    return rule;
  }
}
```

## Файл 4: `bff/src/controllers/auth.controller.ts`

```typescript
import { Controller, Get, Post, Query, Req, Res, Header } from '@nestjs/common';
import { Response, Request } from 'express';
import { AuthService } from '@services/auth.service';
import { JwksService } from '@services/jwks.service';
import { RedisService } from '@services/redis.service';
import { config } from '@configs/configuration';
import { CallbackQueryDto } from '@dto/callback.dto';
import { jwtVerify } from 'jose';
import { Logger } from '../utils/logger';

@Controller()
export class AuthController {
  constructor(
    private authService: AuthService,
    private jwksService: JwksService,
    private redisService: RedisService,
  ) {}

  @Get('login')
  async login(@Req() req: any, @Res() res: Response) {
    const { url } = await this.authService.getAuthorizationUrl(req.session);
    res.redirect(url);
  }

  @Get('callback')
  async callback(
    @Query() query: CallbackQueryDto,
    @Req() req: any,
    @Res() res: Response,
  ) {
    try {
      const { code, state } = query;
      const oldSessionData = {
        accessToken: req.session.accessToken,
        refreshToken: req.session.refreshToken,
        idToken: req.session.idToken,
        userInfo: req.session.userInfo,
      };

      await this.authService.handleCallback(code, state, req.session);

      const newCsrfToken = req.session.csrfToken;
      req.session.regenerate((err: any) => {
        if (err) {
          Logger.error('AuthController', `Session regeneration failed: ${err.message}`);
          res.redirect(`${config.frontendUrl}/login?error=auth_failed`);
          return;
        }

        req.session.accessToken = req.session.accessToken || oldSessionData.accessToken;
        req.session.refreshToken = req.session.refreshToken || oldSessionData.refreshToken;
        req.session.idToken = req.session.idToken || oldSessionData.idToken;
        req.session.userInfo = req.session.userInfo || oldSessionData.userInfo;
        req.session.csrfToken = newCsrfToken;
        req.session.save(() => {
          Logger.info('AuthController', 'Session completed', {
            user: req.session.userInfo?.preferred_username || req.session.userInfo?.email || '?',
          });
          res.redirect(config.frontendUrl + '/');
        });
      });
    } catch (e: any) {
      Logger.error('AuthController', `Callback error: ${e.message}`);
      res.redirect(`${config.frontendUrl}/login?error=auth_failed`);
    }
  }

  @Get('api/me')
  async me(@Req() req: any) {
    return { user: req.session.userInfo };
  }

  @Post('api/auth/refresh')
  async refresh(@Req() req: any, @Res() res: Response) {
    try {
      const tokens = await this.authService.refreshAccessToken(req.session);
      res.json(tokens);
    } catch {
      res.status(401).json({ error: 'Token refresh failed' });
    }
  }

  @Post('logout')
  async logout(@Req() req: any, @Res() res: Response) {
    const idToken = req.session.idToken;
    const userId = req.session.userInfo?.sub;
    const sessionId = req.session.id;
    const refreshToken = req.session.refreshToken;

    if (refreshToken) {
      await this.authService.revokeRefreshToken(refreshToken).catch((err: any) => {
        Logger.warn('AuthController', `Token revocation failed: ${err.message}`);
      });
    }

    if (userId) {
      await this.authService.unregisterSession(userId, sessionId).catch(() => {});
    }

    req.session.destroy((err: any) => {
      if (err) {
        Logger.error('AuthController', `Session destroy error: ${err.message}`);
      }
      res.clearCookie('connect.sid', { path: '/' });
      res.clearCookie('XSRF-TOKEN', { path: '/' });

      if (idToken) {
        res.redirect(this.authService.getLogoutUrl(idToken));
      } else {
        res.redirect('/login');
      }
    });
  }

  @Post('api/auth/backchannel-logout')
  @Header('Cache-Control', 'no-store')
  async backchannelLogout(@Req() req: Request, @Res() res: Response) {
    const logoutToken = req.body?.logout_token;

    if (!logoutToken) {
      res.status(400).send('Missing logout_token');
      return;
    }

    try {
      const { payload } = await jwtVerify(logoutToken, this.jwksService.getJWKS(), {
        issuer: config.keycloak.issuer,
        audience: config.keycloak.clientId,
      });

      const jti = payload.jti as string;
      if (jti) {
        const replayKey = `replay:${jti}`;
        const exists = await this.redisService.client.exists(replayKey);
        if (exists) {
          res.status(401).send('Replay detected');
          return;
        }
        const exp = payload.exp || Math.floor(Date.now() / 1000) + 300;
        const ttl = exp - Math.floor(Date.now() / 1000) + 300;
        await this.redisService.client.set(replayKey, '1', { EX: Math.max(ttl, 60) });
      }

      const sub = payload.sub as string;
      await this.authService.destroyUserSessions(sub);
      res.status(200).send('OK');
    } catch (error: any) {
      Logger.error('AuthController', `Backchannel logout failed: ${error.message}`);
      res.status(401).send('Invalid logout token');
    }
  }
}
```

## Файл 5: `bff/src/controllers/protected_service.controller.ts`

```typescript
import { Controller, Get, Post, Req, Res, Param } from '@nestjs/common';
import { Response } from 'express';
import { httpClient } from '../http/httpClient';
import { config } from '@configs/configuration';
import { Logger } from '../utils/logger';

@Controller('api/service')
export class ProtectedServiceController {
  @Get()
  async proxyGetRoot(@Req() req: any, @Res() res: Response) {
    return this.proxyGet('', req, res);
  }

  @Get('*')
  async proxyGet(@Param('0') path: string, @Req() req: any, @Res() res: Response) {
    try {
      const data = await httpClient.get(`${config.protectedServiceUrl}/${path}`);
      res.json(data);
    } catch (error: any) {
      const status = error.response?.status || 500;
      Logger.error('Proxy', `GET failed: ${error.message}`);
      res.status(status).json({ error: 'Upstream request failed' });
    }
  }

  @Post('*')
  async proxyPost(@Param('0') path: string, @Req() req: any, @Res() res: Response) {
    try {
      const data = await httpClient.post(`${config.protectedServiceUrl}/${path}`, req.body);
      res.json(data);
    } catch (error: any) {
      const status = error.response?.status || 500;
      Logger.error('Proxy', `POST failed: ${error.message}`);
      res.status(status).json({ error: 'Upstream request failed' });
    }
  }
}
```

| Кэширование | `matchRoute()` на каждый запрос | `Map<string, RouteRule>` кэш |

## Как работает

```
GET  /api/service-a/public/docs    → public: true           → 200
POST /api/service-a/orders         → session + csrf          → 200 (if auth)
GET  /api/service-b/data           → roles: ['admin']        → 403 (if user)
POST /api/service-c/items          → client roles writer     → 200 (if has role)
GET  /api/service-d/settings       → roles: all [admin,editor] → 403 (if missing one)
GET  /api/unknown                  → no rule                 → 403 (deny-by-default)
```

## КРИТИЧЕСКАЯ ПРОБЛЕМА: AsyncLocalStorage не работает

### Что происходит

`SessionGuard.canActivate()` делает:
```typescript
return requestContext.run({ session }, () => true);
```

Это создаёт async-контекст ТОЛЬКО внутри `() => true`. Когда `canActivate()` возвращает `true`, контекст завершён. NestJS вызывает handler в оригинальном async-контексте от Express.

**Результат:** `requestContext.getStore()` в axios interceptor всегда возвращает `undefined`. Authorization header никогда не устанавливается.

### Исправление: создать interceptor

Создать `bff/src/http/session-context.interceptor.ts`:

```typescript
import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common';
import { Observable } from 'rxjs';
import { requestContext } from './request-context';

@Injectable()
export class SessionContextInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const req = context.switchToHttp().getRequest();
    const session = req.session;

    if (session?.accessToken) {
      return requestContext.run({ session }, () => next.handle());
    }

    return next.handle();
  }
}
```

Зарегистрировать как глобальный interceptor в `app.module.ts`:
```typescript
import { APP_INTERCEPTOR } from '@nestjs/core';
import { SessionContextInterceptor } from './http/session-context.interceptor';

providers: [
  // ... существующие ...
  { provide: APP_INTERCEPTOR, useClass: SessionContextInterceptor },
],
```

Убрать `requestContext.run()` из `session.guard.ts` (строки 79-83).
