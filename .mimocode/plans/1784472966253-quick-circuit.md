### 4. `client/src/pages/Service.tsx` — Убрать useAuth, грузить данные самому

```tsx
const Service = () => {
  const [serviceData, setServiceData] = useState(null);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    api.get('/api/service')
      .then(res => setServiceData(res.data))
      .catch(err => setError(err.response?.data?.message || 'Failed to fetch'))
      .finally(() => setLoading(false));
  }, []);

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error}</div>;

  return (
    <div>
      <h1>{serviceData?.service || 'Protected Service'}</h1>
      {/* рендер данных из serviceData */}
    </div>
  );
};
```
